===== load-balancer-controller-iam.tf =====
data "aws_iam_policy_document" "load_balancer_controller_trust" {
  statement {
    effect  = "Allow"
    actions = ["sts:AssumeRoleWithWebIdentity"]

    principals {
      type        = "Federated"
      identifiers = [module.eks.oidc_provider_arn]
    }

    condition {
      test     = "StringEquals"
      variable = "${local.application_oidc_host}:aud"
      values   = ["sts.amazonaws.com"]
    }

    condition {
      test     = "StringEquals"
      variable = "${local.application_oidc_host}:sub"
      values = [
        "system:serviceaccount:kube-system:aws-load-balancer-controller"
      ]
    }
  }
}

resource "aws_iam_role" "load_balancer_controller" {
  name        = "streamingapp-capstone-staging-load-balancer-controller"
  description = "IRSA role for the AWS Load Balancer Controller in staging."

  assume_role_policy = data.aws_iam_policy_document.load_balancer_controller_trust.json

  tags = {
    Project     = "streamingapp-capstone"
    Environment = "staging"
    Service     = "aws-load-balancer-controller"
    ManagedBy   = "Terraform"
  }
}

resource "aws_iam_policy" "load_balancer_controller" {
  name        = "streamingapp-capstone-staging-load-balancer-controller"
  description = "AWS Load Balancer Controller v2.14.1 IAM policy."

  policy = file(
    "${path.module}/policies/aws-load-balancer-controller-v2.14.1.json"
  )

  tags = {
    Project     = "streamingapp-capstone"
    Environment = "staging"
    Service     = "aws-load-balancer-controller"
    ManagedBy   = "Terraform"
  }
}

resource "aws_iam_role_policy_attachment" "load_balancer_controller" {
  role       = aws_iam_role.load_balancer_controller.name
  policy_arn = aws_iam_policy.load_balancer_controller.arn
}

output "load_balancer_controller_role_arn" {
  description = "IRSA role ARN for the staging AWS Load Balancer Controller."
  value       = aws_iam_role.load_balancer_controller.arn
}

  application_namespace = "streaming-app-staging"

  application_oidc_host = trimprefix(
    module.eks.cluster_oidc_issuer_url,
    "https://"
  )
}

resource "aws_iam_role" "secret_reader" {
  name               = "streamingapp-capstone-staging-secret-reader"
  description        = "Read the staging application secret through IRSA."
  assume_role_policy = data.aws_iam_policy_document.secret_reader_trust.json

  tags = {
    Service     = "external-secrets"
    Environment = "staging"
  }
}

data "aws_iam_policy_document" "secret_reader_permissions" {
  statement {
    sid    = "ReadStagingApplicationSecret"
    effect = "Allow"

    actions = [
      "secretsmanager:GetSecretValue",
      "secretsmanager:DescribeSecret"
    ]

    resources = [
      aws_secretsmanager_secret.application.arn
    ]
  }
}

resource "aws_iam_role_policy" "secret_reader" {
  name   = "read-staging-application-secret"
  role   = aws_iam_role.secret_reader.id
  policy = data.aws_iam_policy_document.secret_reader_permissions.json
}

output "secret_reader_role_arn" {
  description = "IAM role ARN for the staging secret-reader service account."
  value       = aws_iam_role.secret_reader.arn
}
===== ebs-csi.tf =====

resource "aws_iam_role" "ebs_csi" {
  name        = "streamingapp-capstone-staging-ebs-csi"
  description = "EKS Pod Identity role for the staging EBS CSI driver."

  assume_role_policy = jsonencode({
    Version = "2012-10-17"
    Statement = [
      {
        Effect = "Allow"
        Principal = {
          Service = "pods.eks.amazonaws.com"
        }
        Action = [
          "sts:AssumeRole",
          "sts:TagSession"
        ]
        Condition = {
          StringEquals = {
            "aws:RequestTag/kubernetes-namespace"       = "kube-system"
            "aws:RequestTag/kubernetes-service-account" = "ebs-csi-controller-sa"
          }
        }
      }
    ]
  })

  tags = {
    Project     = "streamingapp-capstone"
    Environment = "staging"
    ManagedBy   = "Terraform"
  }
}

resource "aws_iam_role_policy_attachment" "ebs_csi" {
  role       = aws_iam_role.ebs_csi.name
  policy_arn = "arn:aws:iam::aws:policy/AmazonEBSCSIDriverPolicyV2"
}

resource "aws_eks_addon" "ebs_csi" {
  cluster_name  = module.eks.cluster_name
  addon_name    = "aws-ebs-csi-driver"
  addon_version = "v1.67.2-eksbuild.1"

  resolve_conflicts_on_create = "NONE"
  resolve_conflicts_on_update = "PRESERVE"

  pod_identity_association {
    role_arn        = aws_iam_role.ebs_csi.arn
    service_account = "ebs-csi-controller-sa"
  }

  tags = {
    Project     = "streamingapp-capstone"
    Environment = "staging"
    ManagedBy   = "Terraform"
  }

  depends_on = [
    module.eks,
    aws_iam_role_policy_attachment.ebs_csi
  ]
}
